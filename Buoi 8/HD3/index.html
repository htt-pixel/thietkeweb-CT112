Hoạt động 3: Giải thích luồng chạy mã lệnh (AI Logic Explainer)

Đoạn mã xử lý giảm giá cho thành viên:

let giaGoc = 500000;
let laThanhVien = true;

// Sử dụng toán tử điều kiện (Ternary Operator) thay thế cho if-else rút gọn
let tongTien = laThanhVien ? giaGoc * 0.8 : giaGoc;

console.log("Số tiền cần thanh toán là: " + tongTien);
